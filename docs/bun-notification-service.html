Phase 1.5 — Bun.js Notification Service

Setup bun-notification-service

Panduan setup lengkap service Bun.js yang menjadi consumer Redis Stream, dispatch notifikasi via Firebase Cloud Messaging (HTTP v1 API), dan menyimpan notifikasi in-app ke PostgreSQL.

Bun.js Redis Streams firebase-admin Phase 1.5

💡Mengapa Bun.js

AlasanDetail
SDK FCM resmifirebase-admin Node.js dikelola langsung oleh Google. Support FCM HTTP v1 API penuh. Tidak ada kekhawatiran deprecated seperti Legacy API.
Bun kompatibel dengan package Node.jsfirebase-admin dan ioredis berjalan di Bun tanpa modifikasi. Startup time jauh lebih cepat dari Node.js.
I/O-bound workloadNotification dispatch adalah pure I/O (HTTP ke FCM, write ke DB). Tidak ada CPU-intensive task — Bun cocok untuk ini.
Satu VPS, Docker networkBun.js container berada di Docker Compose network yang sama dengan Rust dan Redis. Tidak ada latency jaringan eksternal antar service.

import admin from "firebase-admin";
import { readFileSync } from "fs";
import { config } from "../config";

// Inisialisasi sekali saat module pertama kali di-import
const serviceAccount = JSON.parse(
  readFileSync(config.firebaseServiceAccount, "utf-8")
);

if (!admin.apps.length) {
  admin.initializeApp({
    credential: admin.credential.cert(serviceAccount),
  });
}

const messaging = admin.messaging();

export async function sendPush(params: {
  token:    string;
  title:    string;
  body:     string;
  deepLink?: string;
  data?:    Record<string, string>;
}): Promise<"ok" | "invalid_token"> {
  try {
    await messaging.send({
      token: params.token,
      notification: { title: params.title, body: params.body },
      data: {
        ...(params.data ?? {}),
        ...(params.deepLink ? { deep_link: params.deepLink } : {}),
      },
      android: { priority: "high" },
      apns: { payload: { aps: { sound: "default" } } },
    });
    return "ok";
  } catch (err: any) {
    // Token tidak valid (unregistered, expired) — jangan retry
    if (
      err?.errorInfo?.code === "messaging/registration-token-not-registered" ||
      err?.errorInfo?.code === "messaging/invalid-registration-token"
    ) {
      return "invalid_token";
    }
    throw err; // error transient — biarkan caller retry
  }
}

🔄Consumer Loop

typescriptbun-notification-service/src/consumer/loop.ts
import Redis from "ioredis";
import { processMessage } from "./processor";
import { config } from "../config";

const STREAM    = "notifications_stream";
const GROUP     = "bun-consumers";
const CONSUMER  = `bun-worker-${process.pid}`;

export async function startConsumerLoop(redis: Redis): Promise<void> {
  // Buat group jika belum ada — idempotent
  try {
    await redis.xgroup("CREATE", STREAM, GROUP, "$", "MKSTREAM");
  } catch (e: any) {
    if (!e.message.includes("BUSYGROUP")) throw e;
  }

  // Recovery: proses PEL (message yang belum di-ACK dari sesi sebelumnya)
  await processPendingEntries(redis);

  console.log(`[consumer] ${CONSUMER} ready — listening on ${STREAM}`);

  // Main loop
  while (true) {
    const results = await redis.xreadgroup(
      "GROUP", GROUP, CONSUMER,
      "COUNT", "10",
      "BLOCK", "2000",
      "STREAMS", STREAM, ">",
    ) as any;

    if (!results) continue; // timeout — loop lagi

    for (const [, messages] of results) {
      for (const [streamId, fields] of messages) {
        await processMessage(redis, STREAM, GROUP, streamId, parseFields(fields));
      }
    }
  }
}

async function processPendingEntries(redis: Redis) {
  const pending = await redis.xautoclaim(
    STREAM, GROUP, CONSUMER,
    60_000, // 60 detik — ambil alih message yang idle > 1 menit
    "0-0", "COUNT", "100",
  ) as any;

  for (const [streamId, fields] of (pending?.[1] ?? [])) {
    await processMessage(redis, STREAM, GROUP, streamId, parseFields(fields));
  }
}

function parseFields(fields: string[]): Record<string, string> {
  const obj: Record<string, string> = {};
  for (let i = 0; i < fields.length; i += 2) obj[fields[i]] = fields[i + 1];
  return obj;
}
typescriptbun-notification-service/src/consumer/processor.ts
import { db } from "../db/client";
import { sendPush } from "../dispatch/fcm";
import { saveInApp } from "../dispatch/inapp";
import { renderTemplate } from "../template";

export async function processMessage(
  redis: any, stream: string, group: string,
  streamId: string, fields: Record<string, string>,
) {
  const { idempotency_key, event_type, recipient_id, actor_name, data } = fields;

  // 1. Idempotency check — INSERT OR IGNORE
  const res = await db.query(
    `INSERT INTO notification.processed_events (idempotency_key)
     VALUES ($1) ON CONFLICT DO NOTHING`,
    [idempotency_key],
  );
  if (res.rowCount === 0) {
    // Sudah diproses — XACK dan skip
    await redis.xack(stream, group, streamId);
    return;
  }

  const parsedData = JSON.parse(data || "{}");
  const { title, body, deepLink } = renderTemplate(event_type, actor_name, parsedData);

  // 2. Dispatch ke semua saluran yang relevan
  const errors: string[] = [];

  try { await saveInApp({ recipient_id, event_type, title, body, deepLink }); }
  catch (e) { errors.push(`inapp: ${e}`); }

  try {
    const tokens = await getDeviceTokens(recipient_id);
    for (const { token, id } of tokens) {
      const result = await sendPush({ token, title, body, deepLink });
      if (result === "invalid_token") await deleteDeviceToken(id);
    }
  } catch (e) { errors.push(`push: ${e}`); }

  // 3. XACK setelah semua saluran selesai
  if (errors.length > 0) {
    // Pindahkan ke DLQ jika ada error
    await redis.xadd("notifications_dlq", "*",
      "original_id", streamId,
      "error", errors.join(" | "),
      ...Object.entries(fields).flat(),
    );
  }
  await redis.xack(stream, group, streamId);
}

📱FCM — HTTP v1 API

✅
firebase-admin menggunakan FCM HTTP v1 API secara otomatis

admin.messaging().send() sudah menggunakan endpoint https://fcm.googleapis.com/v1/projects/{project}/messages:send dengan OAuth2 token yang di-refresh otomatis dari service account. Tidak perlu setup apapun secara manual.

Register Device Token dari Client Mobile

Client mobile (Android/iOS) mendapatkan FCM token via Firebase SDK lalu mengirimkannya ke backend saat login. REST endpoint ini dilayani oleh notification-service Rust.

http
# Register saat login atau token refresh
POST /api/v1/notifications/device-token
Authorization: Bearer {access_token}
Content-Type: application/json

{
  "token":    "fcm-device-token-dari-sdk",
  "platform": "android"   // "android" | "ios" | "web"
}

# Hapus saat logout
DELETE /api/v1/notifications/device-token
Authorization: Bearer {access_token}
ℹ️
Satu user bisa punya multiple device token

User login dari Android + iOS + web browser → tiga token berbeda. Saat push, Bun.js mengirim ke semua token aktif milik user. Token yang kembali dengan status UNREGISTERED dihapus otomatis dari DB.

Bun.js menjalankan SQL migration saat startup sebelum consumer loop dimulai. Menggunakan raw SQL — tidak perlu ORM.

import { readFileSync } from "fs";
import { db } from "./client";

export async function runMigrations() {
  const sql = readFileSync(
    new URL("../../migrations/001_create_notification_tables.sql", import.meta.url),
    "utf-8"
  );
  await db.query(sql);
  console.log("[migrations] notification schema ready");
}
sqlbun-notification-service/migrations/001_create_notification_tables.sql
-- Idempotent: CREATE TABLE IF NOT EXISTS

CREATE TABLE IF NOT EXISTS notification.notifications (
    id           UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    recipient_id UUID        NOT NULL,
    event_type   TEXT        NOT NULL,
    title        TEXT        NOT NULL,
    body         TEXT        NOT NULL,
    deep_link    TEXT,
    is_read      BOOLEAN     NOT NULL DEFAULT false,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_notif_recipient
    ON notification.notifications (recipient_id, created_at DESC);
CREATE INDEX IF NOT EXISTS idx_notif_unread
    ON notification.notifications (recipient_id)
    WHERE is_read = false;

CREATE TABLE IF NOT EXISTS notification.processed_events (
    idempotency_key TEXT        PRIMARY KEY,
    processed_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS notification.device_tokens (
    id         UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id    UUID        NOT NULL,
    token      TEXT        NOT NULL UNIQUE,
    platform   TEXT        NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_device_tokens_user
    ON notification.device_tokens (user_id);

CREATE TABLE IF NOT EXISTS notification.user_preferences (
    user_id       UUID    PRIMARY KEY,
    push_enabled  BOOLEAN NOT NULL DEFAULT true,
    email_enabled BOOLEAN NOT NULL DEFAULT true,
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

❤️Health Check

Bun.js menjalankan HTTP server kecil di port 9000 untuk health check Docker Compose. Consumer loop berjalan di goroutine terpisah.

typescriptbun-notification-service/src/index.ts
import { config } from "./config";     // fail-fast validation
import { createDbClient } from "./db/client";
import { runMigrations } from "./db/migrations";
import { createRedisClient } from "./redis/client";
import { startConsumerLoop } from "./consumer/loop";

async function main() {
  console.log("[startup] bun-notification-service starting...");

  const redis = createRedisClient(config.redisUrl);
  createDbClient(config.databaseUrl);

  // Jalankan migration sebelum consumer dimulai
  await runMigrations();

  // Health check server — tidak blocking
  Bun.serve({
    port: config.port,
    fetch(req) {
      if (new URL(req.url).pathname === "/health") {
        return Response.json({
          status:  "ok",
          service: "bun-notification-service",
          version: "1.0.0",
        });
      }
      return new Response("not found", { status: 404 });
    },
  });
  console.log(`[health] listening on :${config.port}`);

  // Consumer loop — blocking, jalan terus
  await startConsumerLoop(redis);
}

main().catch((e) => { console.error(e); process.exit(1); });
